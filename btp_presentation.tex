% Slide 3: Motivation
\begin{frame}{Motivation: Two Problems, One System}
\begin{columns}[T]
  \column{0.50\textwidth}
  \begin{block}{The Latency Problem}
    \footnotesize
    \begin{itemize}
      \item Payment APIs at $\sim$68{,}000 cross-node ops/sec
      \item 5--15 cross-node calls per request
      \item p95 = 98\,ms, p99 = 285\,ms at peak
      \item Many calls are \textbf{redundant}
    \end{itemize}
  \end{block}
  \begin{exampleblock}{What the Minor Established}
    \footnotesize
    Two-tier in-process cache + 4 safety criteria:
    \textbf{4.1\% network reduction}, \textbf{--11.2\% p95 latency}, zero correctness incidents.
  \end{exampleblock}

  \column{0.48\textwidth}
  \begin{alertblock}{The Concurrency Problem}
    \footnotesize
    Some operations touch \textbf{single, indivisible} resources:
    \begin{itemize}
      \item last inventory unit
      \item unique reservation slot
      \item rate-limited gateway token
      \item account-balance debit
    \end{itemize}
    Caching can re-introduce \emph{stale reads} $\Rightarrow$ double-allocation.
  \end{alertblock}
  \vspace{0.4em}
  \begin{block}{Examiner Question (Minor)}
    \footnotesize\centering
    \emph{``If you cache state, how do you prevent two users from booking the same single piece?''}
  \end{block}
\end{columns}

% Slide 4: Problem Statement
\begin{frame}{Problem Statement}
\begin{block}{We Address}
  \footnotesize
  In a distributed, stateless, high-throughput payment API, how can we
  \begin{enumerate}
    \item safely reduce the number of cross-node calls per request via in-process caching, \textbf{and}
    \item guarantee that concurrent operations on single, indivisible resources cannot result in double-allocation, lost updates, or stale-cache-induced overselling
  \end{enumerate}
  --- without new infrastructure or sacrificing the latency improvements?
\end{block}
\vspace{0.4em}
\begin{columns}[T]
  \column{0.48\textwidth}
  \begin{exampleblock}{Properties Required}
    \footnotesize
    \begin{itemize}
      \item \textbf{Safety:} resource never allocated $>$1 time
      \item \textbf{Liveness:} bounded-time progress
      \item \textbf{Latency budget:} $\le$ 5--10\,ms p95 added
      \item \textbf{Observability:} every outcome auditable
    \end{itemize}
  \end{exampleblock}
  \column{0.48\textwidth}
  \begin{alertblock}{End-Term Contribution}
    \footnotesize
    A \textbf{contention layer} composed of \textbf{five complementary primitives}, plus a decision matrix that selects the right primitive per data class.
  \end{alertblock}
\end{columns}

% Slide 5: Two-Tier Cache Recap
\begin{frame}{Recap: Two-Tier Cache + Safety Boundaries}
\begin{columns}[T]
  \column{0.50\textwidth}
  \begin{block}{Tier 1 --- Thread-Level Cache}
    \footnotesize
    \begin{itemize}
      \item Per-request \texttt{IORef HashMap}
      \item Destroyed on request end
      \item \textbf{Zero staleness}
      \item Hit rate: \textbf{22.3\%}
    \end{itemize}
  \end{block}
  \begin{block}{Tier 2 --- Server-Local Cache}
    \footnotesize
    \begin{itemize}
      \item \texttt{TVar} + LRU + per-class TTL
      \item Shared across threads on a node
      \item TTLs 30--120\,s per data class
      \item Hit rate: \textbf{41.7\%}
    \end{itemize}
  \end{block}

  \column{0.48\textwidth}
  \begin{alertblock}{Cache-Safety Boundaries (must all hold)}
    \footnotesize
    \begin{enumerate}
      \item Immutability within TTL
      \item Request-scope independence
      \item Idempotency of stale reads
      \item Deterministic key
    \end{enumerate}
    Single-resource state \textbf{fails (3)} $\Rightarrow$ explicitly excluded from cache.
  \end{alertblock}
  \begin{exampleblock}{Combined Numbers}
    \footnotesize\centering
    Hit rate \textbf{54.8\%} \,$\bullet$\, calls --4.1\%\\
    p95 --11.2\% \,$\bullet$\, p99 --13.0\%
  \end{exampleblock}
\end{columns}
\end{frame}

% Slide 7: Five primitives
\begin{frame}{The Contention Layer: Five Complementary Primitives}
\begin{columns}[T]
  \column{0.50\textwidth}
  \begin{block}{Primary primitives (one per write)}
    \footnotesize
    \begin{enumerate}
      \item \textbf{Distributed locking} with fencing tokens
      \item \textbf{Atomic decrement} via Redis Lua scripts
      \item \textbf{OCC} with versioned cache entries
      \item \textbf{PN-Counter CRDT} (eventually consistent)
    \end{enumerate}
  \end{block}
  \begin{block}{Composing primitive (always paired)}
    \footnotesize
    \begin{enumerate}\setcounter{enumi}{4}
      \item \textbf{Single-flight} request coalescing
    \end{enumerate}
  \end{block}

  \column{0.48\textwidth}
  \begin{exampleblock}{Why five and not one?}
    \footnotesize
    \begin{itemize}
      \item Hard limit + non-trivial logic $\to$ \textbf{Lock}
      \item Hard limit + simple decrement $\to$ \textbf{Lua}
      \item Low-contention writes $\to$ \textbf{OCC}
      \item Soft holds, over-allocation OK $\to$ \textbf{CRDT}
      \item Reads always pair with $\to$ \textbf{Single-flight}
    \end{itemize}
  \end{exampleblock}
  \begin{alertblock}{Selection tool}
    \footnotesize
    A \textbf{decision matrix} maps each data class to its primary + coalescing primitive (Slide 14).
  \end{alertblock}
\end{columns}
\end{frame}

% Slide 8: Primitive 1 -- Lock + Fence
\begin{frame}[fragile]{Primitive 1 --- Distributed Lock + Fencing Tokens}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Mechanism}
    \footnotesize
    \begin{itemize}
      \item Acquire per-resource lock (Redlock / etcd lease) before read-modify-write.
      \item \textbf{Fence token:} monotonically increasing number; backend rejects stale-token writes.
      \item Mitigates GC pauses, partitions, clock drift.
    \end{itemize}
  \end{block}
\begin{lstlisting}[language=Haskell]
withLock conn key ttl $ \tok -> do
  v  <- backendRead key
  v' <- compute v
  backendWriteIfFence key v' tok
\end{lstlisting}

  \column{0.43\textwidth}
  \begin{exampleblock}{Use when}
    \footnotesize
    Logic non-trivial, contention moderate, critical section calls external services.
  \end{exampleblock}
  \begin{alertblock}{Trade-offs}
    \footnotesize
    \begin{itemize}
      \item +1--3\,ms acquisition latency
      \item Caps per-key throughput
      \item Best for coarse-grained, low-contention resources
    \end{itemize}
  \end{alertblock}
\end{columns}
\end{frame}

% Slide 9: Primitive 2 -- Lua
\begin{frame}[fragile]{Primitive 2 --- Atomic Decrement via Redis Lua}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Mechanism}
    \footnotesize
    Wrap ``check $\geq 1$ + decrement'' in one Lua script. Redis runs scripts atomically.
  \end{block}
\begin{lstlisting}[language={[5.2]Lua}]
local a = tonumber(redis.call('GET', KEYS[1]) or '0')
local w = tonumber(ARGV[1])
if a >= w then
  redis.call('DECRBY', KEYS[1], w)
  return {1, a - w}
else
  return {0, a}
end
\end{lstlisting}

  \column{0.43\textwidth}
  \begin{exampleblock}{Best of all worlds (when applicable)}
    \footnotesize
    \begin{itemize}
      \item Latency: $<$1\,ms
      \item Throughput: $>10^5$ scripts/s/shard
      \item Cache: \emph{display reads} stay cached; authoritative \emph{decrement} bypasses cache + Pub/Sub invalidate
    \end{itemize}
  \end{exampleblock}
  \begin{alertblock}{Limitation}
    \footnotesize
    Critical section must fit in Lua --- no external calls.
  \end{alertblock}
\end{columns}
\end{frame}

% Slide 10: Primitive 3 -- OCC
\begin{frame}[fragile]{Primitive 3 --- OCC with Versioned Cache Entries}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Mechanism}
    \footnotesize
    \begin{itemize}
      \item Each cache entry carries a \texttt{version}.
      \item Reader records observed version.
      \item Writer issues a CAS update conditioned on version unchanged.
      \item Conflict $\Rightarrow$ retry.
    \end{itemize}
  \end{block}
\begin{lstlisting}[language=SQL]
UPDATE inventory
   SET stock = stock - 1, version = version + 1
 WHERE sku = $1 AND version = $2 AND stock >= 1
RETURNING stock, version;
\end{lstlisting}

  \column{0.43\textwidth}
  \begin{exampleblock}{Use when}
    \footnotesize
    Conflicts are \emph{rare} --- e.g., merchant-config writes from the control plane.
  \end{exampleblock}
  \begin{alertblock}{Avoid when}
    \footnotesize
    Contention is high --- retry storms collapse throughput.
  \end{alertblock}
\end{columns}
\end{frame}

% Slide 11: Primitive 4 -- CRDT
\begin{frame}{Primitive 4 --- PN-Counter CRDTs}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Mechanism}
    \footnotesize
    \begin{itemize}
      \item Two grow-only counters per replica: $P_i$ (positive), $N_i$ (negative).
      \item Value $= \sum_i P_i - \sum_i N_i$; merge by element-wise max.
      \item Writes are \textbf{local}, never blocked, eventually converge.
    \end{itemize}
  \end{block}
  \begin{exampleblock}{When to use}
    \footnotesize
    Soft holds, cart-abandonment counters, flash sales advertising ``$\sim$1{,}000 units'' --- where minor overshoot is acceptable.
  \end{exampleblock}

  \column{0.43\textwidth}
  \begin{alertblock}{When NOT to use}
    \footnotesize
    Hard limits where every unit must be exact. CRDTs trade hard upper bound for availability.
  \end{alertblock}
  \begin{block}{Production result}
    \footnotesize
    \textbf{4.7 over-allocations / day} on a 100k-unit base --- under the 1\% business tolerance.
  \end{block}
\end{columns}
\end{frame}

% Slide 12: Primitive 5 -- Single-flight
\begin{frame}{Primitive 5 --- Single-Flight Request Coalescing}
\begin{columns}[T]
  \column{0.55\textwidth}
  \begin{block}{Mechanism}
    \footnotesize
    On a cache miss, only the \textbf{first} concurrent request fetches; the rest wait on the leader's promise and share the result.
  \end{block}
  \begin{exampleblock}{Effect}
    \footnotesize
    \begin{itemize}
      \item Collapses the \emph{front} of contention.
      \item One backend call instead of $k$ identical ones.
      \item Tames cache stampedes / thundering herds.
    \end{itemize}
  \end{exampleblock}

  \column{0.43\textwidth}
  \begin{alertblock}{Important}
    \footnotesize
    Single-flight \textbf{deduplicates on a single node} --- it is \textbf{not} a substitute for atomicity. Compose with Primitive~2 (Lua) for cross-node safety.
  \end{alertblock}
  \begin{block}{Why it matters here}
    \footnotesize
    Reads through the cache always pair with single-flight, so one Redis miss never amplifies into a herd.
  \end{block}
\end{columns}
\end{frame}
